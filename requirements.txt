# ============ HachiYang 依赖清单 ============
# Python >= 3.11
# 一键安装：pip install -r requirements.txt
# 如需锁定精确版本：pip freeze > requirements.lock.txt

# ---- Phase 0 必需：基础设施 ----
openai                 # 火山方舟 LLM/Embedding（OpenAI 兼容端点）
pydantic               # 核心数据结构（core/models.py）
pydantic-settings      # 配置加载与校验（config.py）
python-dotenv          # .env 加载（pydantic-settings env_file 依赖）
chromadb               # 向量库（rag_knowledge / episodic_memory）

# ---- Phase 1：飞书渠道 + Web 服务 ----
fastapi                # 飞书 webhook 回调 + 本地调试台
uvicorn[standard]      # ASGI 服务器
lark-oapi              # 飞书官方 SDK

# ---- Phase 3：RAG 文档解析 ----
EbookLib               # EPUB 解析
beautifulsoup4         # EPUB HTML 解析
PyMuPDF                # PDF 解析（Outline + 页文本 + 术语索引页）

# ---- Phase 7：测试 ----
pytest                 # 单元测试 + 集成测试

# ---- 可选 ----
jieba                # FTS5 中文分词增强（非强制，需要时取消注释）
